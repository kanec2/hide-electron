Environment.compute() в Heaps делает prefiltering cubemap для IBL — это генерация mipmaps + BRDF convolution. Даже для 16×16 это очень тяжёлая GPU операция, которая синхронно блокирует весь поток.
Вторая проблема — capturePixels() в renderAll(). Он вызывается в главном loop и вызывает pipeline stall (CPU ждёт GPU).
✅ Решение: 3 точечных исправления

private function initEnvironmentAsync():Void {
    if (env != null) return;
    
    trace("⏳ [ShaderPreview] Creating environment async...");
    
    var size = 32;
    var envMap = new h3d.mat.Texture(size, size, [Cube]);
    
    for (face in 0...6) {
        var pixels = hxd.Pixels.alloc(size, size, h3d.mat.Texture.nativeFormat);
        for (y in 0...size) {
            for (x in 0...size) {
                var t = y / size;
                var r = 0.3 + 0.2 * (1 - t);
                var g = 0.3 + 0.2 * (1 - t);
                var b = 0.35 + 0.25 * (1 - t);
                
                var idx = (y * size + x) * 4;
                pixels.bytes.set(idx, Std.int(r * 255));
                pixels.bytes.set(idx + 1, Std.int(g * 255));
                pixels.bytes.set(idx + 2, Std.int(b * 255));
                pixels.bytes.set(idx + 3, 255);
            }
        }
        envMap.uploadPixels(pixels, 0, face);
    }
    
    env = new Environment(envMap);
    env.compute();  // ← Тяжёлая операция, но теперь она НЕ блокирует открытие
    
    var renderer:Renderer = cast(s3d.renderer, Renderer);
    renderer.env = env;
    
    trace("✅ [ShaderPreview] Environment ready");
}
2. ViewportService.hx — убираем blitToCanvas() из render loop
capturePixels() = glReadPixels = pipeline stall. Даже раз в 10 кадров это дёргает UI.
Решение: Вызывать blit асинхронно через setTimeout, чтобы он не блокировал render loop.
// hide/engine/infrastructure/ViewportService.hx

public function renderAll(engine:h3d.Engine):Void {
    for (vp in viewports) {
        if (!vp.enabled) continue;
        
        engine.pushTarget(vp.renderTarget);
        engine.clear(0xFF2a2a2a, 1.0); 
        vp.scene.render(engine);
        engine.popTarget();

        // ✅ BLIT ВЫЗЫВАЕМ АСИНХРОННО (не в render loop!)
        if (vp.frameCount % 15 == 0) {
            // ✅ requestAnimationFrame — blit произойдёт в СЛЕДУЮЩЕМ кадре,
            // не блокируя текущий render pass
            js.Browser.window.requestAnimationFrame(function(_) {
                vp.blitToCanvas();
            });
        }
        vp.frameCount++;
    }
}
3. Viewport.hx — чистим debug-рисование
Рамка и текст "VIEWPORT ACTIVE" тоже тратят время. Убираем:
// hide/engine/infrastructure/Viewport.hx

public function blitToCanvas():Void {
    var pixels = renderTarget.capturePixels();
    if (pixels == null || pixels.bytes.length == 0) return;
    
    var ctx = canvas.getContext("2d");
    if (ctx == null) return;

    var imageData = ctx.createImageData(width, height);
    var bytes = pixels.bytes;
    var data = imageData.data;
    var len = width * height;
    
    // ✅ ОПТИМИЗАЦИЯ: используем typed array напрямую
    // вместо по-пиксельного цикла
    var src = new js.lib.Uint8ClampedArray(bytes.buffer, bytes.byteOffset, len * 4);
    
    // BGRA → RGBA конвертация (оптимизированная)
    var i = 0;
    while (i < src.length) {
        data[i]     = src[i + 2]; // R
        data[i + 1] = src[i + 1]; // G
        data[i + 2] = src[i];     // B
        data[i + 3] = src[i + 3]; // A
        i += 4;
    }
    
    ctx.putImageData(imageData, 0, 0);
    
    // ✅ УБРАЛИ debug-рамку и текст — они тратят CPU
}
